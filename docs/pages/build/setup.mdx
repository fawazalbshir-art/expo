---import React, { useState, useEffect } from "react"; import { View, Text, TextInput, ScrollView, TouchableOpacity, Alert, FlatList, } from "react-native"; import * as SQLite from "expo-sqlite";

const db = SQLite.openDatabase("clinic.db");

export default function App() { const [screen, setScreen] = useState("list");

const [name, setName] = useState(""); const [age, setAge] = useState(""); const [phone, setPhone] = useState(""); const [address, setAddress] = useState(""); const [complaint, setComplaint] = useState(""); const [history, setHistory] = useState(""); const [treatment, setTreatment] = useState("");

const [patients, setPatients] = useState([]); const [search, setSearch] = useState("");

useEffect(() => { db.transaction((tx) => { tx.executeSql( "CREATE TABLE IF NOT EXISTS patients (id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT, age TEXT, phone TEXT, address TEXT, complaint TEXT, history TEXT, treatment TEXT, date TEXT);" ); }); loadPatients(); }, []);

const loadPatients = () => { db.transaction((tx) => { tx.executeSql("SELECT * FROM patients", [], (_, { rows }) => { setPatients(rows._array); }); }); };

const savePatient = () => { if (!name) return Alert.alert("خطأ", "أدخل اسم المريض");

db.transaction((tx) => {
  tx.executeSql(
    "INSERT INTO patients (name, age, phone, address, complaint, history, treatment, date) VALUES (?, ?, ?, ?, ?, ?, ?, ?)",
    [name, age, phone, address, complaint, history, treatment, new Date().toISOString()],
    () => {
      Alert.alert("تم", "تم حفظ المريض بنجاح");
      clearForm();
      loadPatients();
      setScreen("list");
    }
  );
});

};

const clearForm = () => { setName(""); setAge(""); setPhone(""); setAddress(""); setComplaint(""); setHistory(""); setTreatment(""); };

const filtered = patients.filter((p) => p.name?.includes(search) );

const renderList = () => ( <View style={{ flex: 1 }}> <Text style={{ fontSize: 22, fontWeight: "bold", marginBottom: 10 }}> عيادة الدكتور فواز عبدالرحمن البشير </Text>

<TextInput
    placeholder="بحث عن مريض"
    value={search}
    onChangeText={setSearch}
    style={{ backgroundColor: "white", padding: 10, borderRadius: 10, marginBottom: 10 }}
  />

  <TouchableOpacity
    onPress={() => setScreen("add")}
    style={{ backgroundColor: "#2563eb", padding: 12, borderRadius: 10, marginBottom: 10 }}
  >
    <Text style={{ color: "white", textAlign: "center" }}>إضافة مريض جديد</Text>
  </TouchableOpacity>

  <FlatList
    data={filtered}
    keyExtractor={(item) => item.id.toString()}
    renderItem={({ item }) => (
      <View style={{ backgroundColor: "white", padding: 12, borderRadius: 10, marginBottom: 10 }}>
        <Text style={{ fontWeight: "bold" }}>{item.name}</Text>
        <Text>{item.phone}</Text>
        <Text>{item.date}</Text>
      </View>
    )}
  />
</View>

);

const renderAdd = () => ( <ScrollView> <Text style={{ fontSize: 20, fontWeight: "bold", marginBottom: 10 }}> إضافة مريض جديد </Text>

{[
    [name, setName, "اسم المريض"],
    [age, setAge, "العمر"],
    [phone, setPhone, "رقم الهاتف"],
    [address, setAddress, "العنوان"],
  ].map(([val, setVal, ph], i) => (
    <TextInput
      key={i}
      placeholder={ph}
      value={val}
      onChangeText={setVal}
      style={{ backgroundColor: "white", padding: 10, borderRadius: 10, marginBottom: 10 }}
    />
  ))}

  <TextInput placeholder="الشكوى" value={complaint} onChangeText={setComplaint}
    style={{ backgroundColor: "white", padding: 10, borderRadius: 10, marginBottom: 10, height: 80 }} multiline />

  <TextInput placeholder="التاريخ المرضي" value={history} onChangeText={setHistory}
    style={{ backgroundColor: "white", padding: 10, borderRadius: 10, marginBottom: 10, height: 80 }} multiline />

  <TextInput placeholder="العلاج" value={treatment} onChangeText={setTreatment}
    style={{ backgroundColor: "white", padding: 10, borderRadius: 10, marginBottom: 10, height: 80 }} multiline />

  <TouchableOpacity onPress={savePatient} style={{ backgroundColor: "green", padding: 15, borderRadius: 10 }}>
    <Text style={{ color: "white", textAlign: "center" }}>حفظ المريض</Text>
  </TouchableOpacity>

  <TouchableOpacity onPress={() => setScreen("list")} style={{ marginTop: 10 }}>
    <Text style={{ textAlign: "center", color: "blue" }}>رجوع</Text>
  </TouchableOpacity>
</ScrollView>

);

return ( <View style={{ flex: 1, padding: 20, paddingTop: 50 }}> {screen === "list" ? renderList() : renderAdd()} </View> ); }
